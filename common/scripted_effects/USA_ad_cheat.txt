# opened by the focus "Question the Count" (or event usa_ad.20 pre)
USA_ad_mid_open = {
	set_variable = { usa_midterm_cheat = 0 }
	set_country_flag = usa_midterms_open
	log = "[GetDateText]: American Dream - midterms window open"
}

# small shared helper: burn legitimacy -> some of it becomes chaos
USA_ad_burn_legit = {			# set temp burn = points
	set_temp_variable = { legit_delta = burn }
	multiply_temp_variable = { legit_delta = -1 }
	USA_ad_legit_add = yes
	set_temp_variable = { chaos_delta = burn }
	multiply_temp_variable = { chaos_delta = 0.3 }		# 30% of the burn leaks to chaos now
	USA_chaos_add = yes
}

# --- resolution ---
# Certify Our Result. Available once you have cheated enough.
# It locks in control and hands off to the Blatant/Quiet CHOICE event
# (usa_ad.62), whose option completes the matching focus - that focus's
# reward carries the big chaos beat. So the beat fires exactly once.
USA_ad_mid_certify = {
	clr_country_flag = usa_midterms_open
	set_country_flag = usa_stole_midterms

	# lock in GOP/MAGA control no matter what the count said
	if = {
		limit = { check_variable = { var = usa_house_gop value = usa_house_dem compare = less_than_or_equals } }
		set_variable = { usa_house_gop = 224 }
		set_variable = { usa_house_dem = 211 }
	}

	set_temp_variable = { legit_delta = -6 }
	USA_ad_legit_add = yes

	country_event = { id = usa_ad.62 hours = 4 }		# "The Count Is Ours" - Blatant / Quiet
	USA_ad_politics_recalc = yes
}
