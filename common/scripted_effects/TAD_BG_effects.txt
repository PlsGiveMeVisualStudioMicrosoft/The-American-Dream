TAD_BG_refresh_BLZ_state_modifiers = {
    15 = { TAD_BG_clear_BLZ_state_modifiers = yes }
    88 = { TAD_BG_clear_BLZ_state_modifiers = yes }

    # Defence network: exactly one tier.
    if = {
        limit = {
            OR = {
                has_country_flag = BLZ_total_defence_activated
                has_country_flag = BLZ_last_line_activated
                has_country_flag = BLZ_GUA_war_started
                has_completed_focus = BLZ_total_defence_focus
                has_completed_focus = BLZ_hold_until_world_listens
                has_completed_focus = BLZ_national_survival_plan
            }
        }
        15 = { add_dynamic_modifier = { modifier = BLZ_defence_network_tier_3 } }
        88 = { add_dynamic_modifier = { modifier = BLZ_defence_network_tier_3 } }
    }
    else_if = {
        limit = {
            OR = {
                has_country_flag = BLZ_defence_network_improved
                has_completed_focus = BLZ_prepare_belmopan
                has_completed_focus = BLZ_jungle_defence_training
                has_completed_focus = BLZ_jungle_warfare_school
                has_completed_focus = BLZ_civil_defence_committees
                has_completed_focus = BLZ_integrated_defence_command
            }
        }
        15 = { add_dynamic_modifier = { modifier = BLZ_defence_network_tier_2 } }
        88 = { add_dynamic_modifier = { modifier = BLZ_defence_network_tier_2 } }
    }
    else = {
        15 = { add_dynamic_modifier = { modifier = BLZ_defence_network_tier_1 } }
        88 = { add_dynamic_modifier = { modifier = BLZ_defence_network_tier_1 } }
    }

    # Mopan line: zero or one tier.
    if = {
        limit = {
            OR = {
                has_country_flag = BLZ_mopan_line_advanced
                has_completed_focus = BLZ_preemptive_border_garrison
                has_completed_focus = BLZ_western_highway_redoubts
            }
        }
        15 = { add_dynamic_modifier = { modifier = BLZ_mopan_line_tier_2 } }
        88 = { add_dynamic_modifier = { modifier = BLZ_mopan_line_tier_2 } }
    }
    else_if = {
        limit = {
            OR = {
                has_country_flag = BLZ_mopan_line_built
                has_completed_focus = BLZ_fortify_mopan_border
                has_completed_focus = BLZ_prepare_belmopan
            }
        }
        15 = { add_dynamic_modifier = { modifier = BLZ_mopan_line_tier_1 } }
        88 = { add_dynamic_modifier = { modifier = BLZ_mopan_line_tier_1 } }
    }

    # Distributed supply network: zero or one tier.
    if = {
        limit = {
            OR = {
                has_country_flag = BLZ_supply_network_advanced
                has_completed_focus = BLZ_emergency_radio_network
                has_completed_focus = BLZ_civilian_logistics_corps
                has_completed_focus = BLZ_integrated_defence_command
            }
        }
        15 = { add_dynamic_modifier = { modifier = BLZ_supply_network_tier_2 } }
        88 = { add_dynamic_modifier = { modifier = BLZ_supply_network_tier_2 } }
    }
    else_if = {
        limit = {
            OR = {
                has_country_flag = BLZ_supply_network_started
                has_completed_focus = BLZ_disperse_reserve_stocks
                has_completed_focus = BLZ_emergency_radio_network
            }
        }
        15 = { add_dynamic_modifier = { modifier = BLZ_supply_network_tier_1 } }
        88 = { add_dynamic_modifier = { modifier = BLZ_supply_network_tier_1 } }
    }
}
