# ==============================================================
#  THE AMERICAN DREAM - UNITED STATES CONGRESS  (effects)
#
#  Draws the House / Senate hemicycle window from a small set of
#  country variables. The diagram is a PURE VISUALISER - it never
#  owns the numbers, it only reads them:
#
#    usa_house_gop  usa_house_dem   (must sum to 435)
#    usa_senate_gop usa_senate_dem  (must sum to 100)
#
#  Everything is driven from common/on_actions/USA_congress_on_actions.txt
#  via USA_congress_diagram_tick, which fires at startup and daily.
#  That means the diagram works no matter WHO raises the
#  has_country_flag = usa_ad_initialized switch - the mod's own
#  Congress/tribes init, or the fallback seeder below for a bare
#  install. It is deliberately NOT gated behind that flag being set
#  by this file, which was the reason the boxes came up empty.
#
#  Party ids double as sprite frames in GFX_usa_congress_party_icon:
#         1 = GOP  (red frame 1)     4 = Dem  (blue frame 4)
#
#  ---- Writing to Congress from your own content ----
#  Never touch usa_*_main_array directly. Move seats through the
#  named variables; the daily tick repaints, or call it now:
#         add_to_variable      = { usa_house_gop = 3 }
#         subtract_from_variable = { usa_house_dem = 3 }
#         USA_congress_refresh = yes
# ==============================================================

# --------------------------------------------------------------
#  MAIN TICK  (called from on_startup and on_daily_USA)
#
#  1. If nothing has initialised Congress yet, seed sensible
#     defaults - but only AFTER inauguration, so we can never race
#     an external init that runs at game start.
#  2. Once the system is live (however it was raised), lay out the
#     static dot positions once, then repaint the seats.
# --------------------------------------------------------------
USA_congress_diagram_tick = {
	# Seed sensible defaults the moment nothing else has set Congress
	# up. (If your own Congress/tribes init should own these numbers,
	# gate this behind a flag your init sets - or just tell me and I'll
	# do it. The daily repaint reads the live variables either way.)
	if = {
		limit = { NOT = { has_country_flag = usa_ad_initialized } }
		USA_congress_seed_defaults = yes
	}

	# Lay out the static dot positions once...
	if = {
		limit = { NOT = { has_country_flag = usa_congress_positions_built } }
		USA_house_positions = yes
		USA_senate_positions = yes
		set_country_flag = usa_congress_positions_built
	}

	# ...then repaint the seats from whatever the variables are now.
	# Deliberately NOT gated on usa_ad_initialized, so the diagram
	# fills in no matter how the numbers got set.
	USA_congress_refresh = yes
}

# --------------------------------------------------------------
#  FALLBACK SEEDER  (only fires on a bare install with no other
#  Congress system - edit the starting numbers freely)
# --------------------------------------------------------------
USA_congress_seed_defaults = {
	# House: 435 seats total (GOP + Dem must sum to 435)
	set_variable = { usa_house_gop = 220 }
	set_variable = { usa_house_dem = 215 }
	# Senate: 100 seats total (GOP + Dem must sum to 100)
	set_variable = { usa_senate_gop = 53 }
	set_variable = { usa_senate_dem = 47 }

	# Tribes - overlapping head-counts, a member can be several.
	# MAGA super-majority triggers at 290 (House) / 67 (Senate).
	set_variable = { usa_house_maga   = 160 }
	set_variable = { usa_senate_maga  = 30 }
	set_variable = { usa_house_prowar = 250 }
	set_variable = { usa_senate_prowar = 68 }
	set_variable = { usa_house_estab  = 300 }
	set_variable = { usa_senate_estab = 82 }

	# Union legitimacy - the union starts to fracture below 30.
	set_variable = { usa_legitimacy = 72 }

	set_country_flag = usa_ad_initialized
	log = "[GetDateText]: USA Congress defaults seeded (fallback)"
}

# --------------------------------------------------------------
#  REPAINT BOTH CHAMBERS
# --------------------------------------------------------------
USA_congress_refresh = {
	USA_house_reload = yes
	USA_senate_reload = yes
}

# --------------------------------------------------------------
#  REBUILD ONE CHAMBER'S SEAT ARRAY FROM ITS GOP/DEM VARIABLES
#
#  usa_*_main_array holds one entry per seat: the party id of the
#  seat's occupant. We lay the GOP block down first (ids = 1) then
#  the Dem block (ids = 4), one add_to_array per seat. The x/y
#  arrays are ordered left -> right, so GOP fills the left wedge
#  and Dem the right. (Plain add_to_array in a while loop - no
#  resize_array - so the array is always exactly GOP+Dem long.)
# --------------------------------------------------------------
USA_house_reload = {
	clear_array = usa_house_main_array

	set_temp_variable = { seat_i = 0 }
	while_loop_effect = {
		limit = { check_variable = { seat_i < usa_house_gop } }
		add_to_array = { usa_house_main_array = 1 }   # GOP
		add_to_temp_variable = { seat_i = 1 }
	}

	set_temp_variable = { seat_i = 0 }
	while_loop_effect = {
		limit = { check_variable = { seat_i < usa_house_dem } }
		add_to_array = { usa_house_main_array = 4 }   # Dem
		add_to_temp_variable = { seat_i = 1 }
	}
	log = "[GetDateText]: House repainted - GOP [?usa_house_gop] Dem [?usa_house_dem]"
}

USA_senate_reload = {
	clear_array = usa_senate_main_array

	set_temp_variable = { seat_i = 0 }
	while_loop_effect = {
		limit = { check_variable = { seat_i < usa_senate_gop } }
		add_to_array = { usa_senate_main_array = 1 }   # GOP
		add_to_temp_variable = { seat_i = 1 }
	}

	set_temp_variable = { seat_i = 0 }
	while_loop_effect = {
		limit = { check_variable = { seat_i < usa_senate_dem } }
		add_to_array = { usa_senate_main_array = 4 }   # Dem
		add_to_temp_variable = { seat_i = 1 }
	}
	log = "[GetDateText]: Senate repainted - GOP [?usa_senate_gop] Dem [?usa_senate_dem]"
}
