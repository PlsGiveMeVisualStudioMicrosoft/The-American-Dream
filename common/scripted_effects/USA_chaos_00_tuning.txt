# ==============================================================
#  USA CHAOS SYSTEM - TUNING
#
#  Chaos now runs on the SAME kind of leash as approval. It
#  follows a 48-month target curve inside a band; push it out of
#  the band with an event or decision and it bleeds back to the
#  edge, so a spike is temporary instead of permanent.
#
#  THIS IS THE ONLY CHAOS FILE YOU NEED TO EDIT.
#  Everything else is machinery and can be left alone.
#
#  Two things live here:
#     1. USA_chaos_load_tuning - the knobs
#     2. USA_chaos_load_curve   - the 48 monthly target numbers
#
#  Both are re-read every time the leash is armed.
#
#  ONE-TIME SWITCH IN THE APPROVAL TUNING FILE:
#     In USA_approval_00_tuning.txt set
#         usa_chaos_creep_cap = 0
#     to retire the old upward "creep". The curve below replaces
#     it. (Leaving creep on just adds a tiny bias on top; 0 is
#     cleanest.)
# ==============================================================


USA_chaos_load_tuning = {

	# ---- THE LEASH ------------------------------------------
	# usa_chaos_band  the corridor. Chaos is free to sit anywhere
	#                 within this many points of the month's
	#                 target. Outside it, the daily bleed walks it
	#                 back. Lower = tighter rails.
	#
	# usa_chaos_pull  monthly decay INSIDE the band only. 0.5 =
	#                 half the remaining gap to target per month.
	#                 Skipped while outside the band, because the
	#                 bleed already owns the correction there.
	#
	# usa_chaos_bleed_per_day
	#                 how fast chaos walks back to the edge of the
	#                 band, per day, in real points. 0.5 reads on
	#                 the bar as -1 every two days. Raise for a
	#                 harder correction after a spike, lower for a
	#                 softer one.

	set_variable = { usa_chaos_band = 8 }
	set_variable = { usa_chaos_pull = 0.5 }
	set_variable = { usa_chaos_bleed_per_day = 0.5 }

	# ---- NO OVERSHOOT EVENTS --------------------------------
	# Unlike approval, chaos is a SILENT corridor: it bleeds back
	# with no "why did that happen" pop-up. The only chaos-flavour
	# news comes from the scheduled news events in the event file.
	# (If you ever want overshoot events, copy the high/low bag
	#  machinery from the approval system and add the two knobs
	#  usa_chaos_event_at / usa_chaos_evt_cooldown here.)
}
