BLZ_recruit_caribbean_manpower_pool = {
    icon = GFX_decision_generic_propaganda
    allowed = { tag = BLZ }
    visible = { has_completed_focus = BLZ_proclaim_emergency_defense_act }
    available = {
        check_variable = { BLZ_GUA_tension > 39 }
        NOT = { has_country_flag = BLZ_used_caribbean_manpower_pool }
    }
    cost = 55
    fire_only_once = yes
    complete_effect = {
        set_country_flag = BLZ_used_caribbean_manpower_pool
        add_manpower = 700
        add_timed_idea = { idea = BLZ_caribbean_manpower_contract days = 180 }
        add_to_variable = { BLZ_international_support = 2 }
        TAD_BG_clamp_metrics = yes
    }
    ai_will_do = {
        base = 35
        modifier = { factor = 4 has_war_with = GUA }
        modifier = { factor = 4 has_manpower < 1000 }
    }
}
