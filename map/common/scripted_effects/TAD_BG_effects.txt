TAD_BG_refresh_BLZ_state_modifiers = {
	remove_ideas = BLZ_homeland_defence_doctrine
	remove_ideas = BLZ_belmopan_fortified
	remove_ideas = BLZ_jungle_defence_network
	remove_ideas = BLZ_total_defence
	remove_ideas = BLZ_last_line_of_defence
	TAD_BG_apply_initial_BLZ_state_modifiers = yes
	if = { limit = { has_completed_focus = BLZ_prepare_belmopan } TAD_BG_apply_belmopan_fortification = yes }
	if = { limit = { OR = { has_completed_focus = BLZ_jungle_defence_training has_completed_focus = BLZ_jungle_warfare_school } } TAD_BG_apply_jungle_defence = yes }
	if = { limit = { has_completed_focus = BLZ_total_defence_focus } TAD_BG_apply_total_defence = yes }
	if = { limit = { OR = { has_completed_focus = BLZ_hold_until_world_listens has_country_flag = BLZ_GUA_war_started } } TAD_BG_apply_last_line = yes }
	if = { limit = { has_completed_focus = BLZ_civil_defence_committees } TAD_BG_apply_civil_defence = yes }
	if = { limit = { has_completed_focus = BLZ_disperse_reserve_stocks } TAD_BG_apply_distributed_stockpiles = yes }
	if = { limit = { has_completed_focus = BLZ_emergency_radio_network } TAD_BG_apply_emergency_comms = yes }
	if = { limit = { has_completed_focus = BLZ_hold_the_sarstoon } TAD_BG_apply_sarstoon_denial = yes }
	if = { limit = { has_completed_focus = BLZ_continuity_of_government } TAD_BG_apply_continuity = yes }
}

TAD_BG_refresh_GUA_state_modifiers = {
	remove_ideas = GUA_peten_logistical_bottleneck
	remove_ideas = GUA_peten_logistics_improved
	remove_ideas = GUA_forward_supply_network
	if = {
		limit = { OR = { has_completed_focus = GUA_forward_supply_depots has_country_flag = GUA_forward_depots_built } }
		TAD_BG_set_peten_forward_network = yes
	}
	else_if = {
		limit = { has_completed_focus = GUA_peten_logistics_corps }
		TAD_BG_set_peten_logistics_improved = yes
	}
	else = { 89 = { add_dynamic_modifier = { modifier = GUA_peten_bottleneck_state_modifier } } }
}
